class User {
  - id : int
  - name : string
  - email : string
  - password : string
  - role : string
  - created_at : datetime
  + login(email, pass) : boolean
  + logout() : void
  + resetPassword(newPass) : boolean
}

class Siswa {
  - id : int
  - user_id : int
  - kelas_id : int
  - nisn : string
  - nama_lengkap : string
  - jenkel : string
  - alamat : text
  + getDetailSiswa() : object
  + getRiwayatTagihan() : list<Tagihan>
}

class Kelas {
  - id : int
  - tingkat_kelas : string
  - nama_kelas : string
  - wali_kelas : string
  + getDaftarSiswa() : list<Siswa>
}

class KategoriTagihan {
  - id : int
  - nama_kategori : string
  - nominal_default : double
  - tipe_pembayaran : string
  + createKategori() : boolean
  + ubahNominal(newNominal) : void
}

class Tagihan {
  - id : int
  - siswa_id : int
  - kategori_id : int
  - nominal_tertagih : double
  - batas_waktu : date
  - status : string
  + generateMassal(kelas_id, kat_id) : boolean
  + cekTunggakanActive(siswa_id) : list<Tagihan>
  + getSisaHutang() : double
}

class Pembayaran {
  - id : int
  - tagihan_id : int
  - verified_by : int
  - tgl_bayar : datetime
  - nominal_dibayar : double
  - file_bukti : string
  - catatan : text
  + unggahBukti(fileStream) : boolean
  + setujuiPembayaran(bendahara_id) : boolean
  + tolakPembayaran(alasan_tolak) : void
  + cetakKwitansiLunas() : PDFDocument
}

class Pengeluaran {
  - id : int
  - author_id : int
  - tgl_pengeluaran : date
  - deskripsi_keperluan : string
  - nominal_keluar : double
  - nota_lampiran : string
  + catatKasKeluar(data) : boolean
  + hitungTotalArusKasKeluar(start, end) : double
}
